{
    uint32_t aw = (uint32_t)a->width, bw = (uint32_t)b->width;
    uint32_t ah = (uint32_t)a->height, bh = (uint32_t)b->height;

    if (aw < bw) {
        return -1;
    }
    if (aw <= bw) {
        if (ah < bh) {
            return -1;
        }
        if (ah <= bh) {
            return 0;
        }
    }
    return 1;
}
