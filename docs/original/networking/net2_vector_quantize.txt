    cos_yaw = (real)cos((double)yaw);
    cos_pitch = (real)cos((double)pitch);

        sin_yaw = 0.0f;

    if (sin_pitch < 0.0001f && -0.0001f < sin_pitch) {
        sin_pitch = 0.0f;
